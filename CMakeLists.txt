cmake_minimum_required(VERSION 3.16)
project(ReliableExpenseTracker VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(expense_core src/Expense.cpp src/ExpenseTracker.cpp)
target_include_directories(expense_core PUBLIC include)
add_executable(expense_tracker src/main.cpp)
target_link_libraries(expense_tracker PRIVATE expense_core)

function(expense_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endfunction()
expense_warnings(expense_core)
expense_warnings(expense_tracker)

include(CTest)
if(BUILD_TESTING)
    add_executable(expense_tests tests/test_core.cpp)
    target_link_libraries(expense_tests PRIVATE expense_core)
    expense_warnings(expense_tests)
    add_test(NAME expense_core COMMAND expense_tests)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME expense_cli COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:expense_tracker>)
    set_tests_properties(expense_core expense_cli PROPERTIES TIMEOUT 60)
endif()

install(TARGETS expense_tracker RUNTIME DESTINATION bin)
